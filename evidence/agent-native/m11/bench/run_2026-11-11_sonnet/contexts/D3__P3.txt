    92 |     if (ev.text.len <= w) return null;
    93 |     if (sp.end > ev.text.len) return null;
    94 |     if (!std.mem.eql(u8, ev.text, ev.line_bytes)) return null;
    95 |     const m0 = sp.start;
    96 |     const m1 = sp.end;
    97 |     var start: usize = undefined;
    98 |     var end: usize = undefined;
    99 |     if (m1 - m0 >= w) {
   100 |         // The match alone does not fit: the window is the match. Nothing is
   101 |         // gained, but the "never cut the match" rule holds unconditionally.
   102 |         start = m0;
   103 |         end = m1;
   104 |     } else {
   105 |         // Centre the budget on the match and clamp to the line.
   106 |         const pad = (w - (m1 - m0)) / 2;
   107 |         const s0 = if (m0 > pad) m0 - pad else 0;
   108 |         var s = s0;
   109 |         var e = @min(ev.text.len, s0 + w);
   110 |         s = if (e > w) e - w else 0;
   111 |         // Align to codepoint boundaries without ever entering the match.
   112 |         while (s < m0 and (ev.text[s] & 0xC0) == 0x80) s += 1;
   113 |         // `e` is an exclusive end: at `e == text.len` the window already ends
   114 |         // on the line's last byte, and `text[e]` would read past the slice
   115 |         // (panic in Debug/ReleaseSafe, undefined behaviour in ReleaseFast).

   117 |         start = s;
   118 |         end = e;
   119 |     }
