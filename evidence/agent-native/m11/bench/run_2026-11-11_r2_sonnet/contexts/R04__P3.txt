    21 | ///   `\?`  ->   `?`          zero or one
    22 | ///   `(`   ->   `\(`         literal parenthesis
    23 | ///   `)`   ->   `\)`
    24 | ///   `{`   ->   `\{`         literal brace
    25 | ///   `}`   ->   `\}`
    26 | ///   `|`   ->   `\|`         literal pipe
    27 | ///   `+`   ->   `\+`         literal plus
    28 | ///   `?`   ->   `\?`         literal question
